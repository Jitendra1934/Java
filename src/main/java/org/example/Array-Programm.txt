//Count the occurece of each character in given Array


import java.util.Arrays;
class Main {
    public static void main(String[] args) {
       
      int[] a = {1,2,4,5,3,2,5,2,5,6,7,3,5,7,4,5,5,6};
       
      int[] v  = new int[a.length];
      
      int vis = -1;
     
      for(int i =0; i<a.length; i++){
          
        int count = 1;
        
        for(int j = i+1; j<a.length; j++){
            
            if(a[i] == a[j]){
                
                count++;
                
                v[j] = vis;
            }
            if(v[i]!=vis){
                
                v[i]=count;
            }
        }
        
      }
      for(int i=0; i<v.length; i++){
          if(v[i]!=vis){
              System.out.println(a[i]+" Ocuures "+v[i]+" times");
          }
      }
       
    }
}

//  Rigth rotation of an array


class Main {
    public static void main(String[] args) {
        
        int[]  a ={1,2,3,4,5};
        
        int n=3;
        
        for(int i=0; i<n; i++){
            
            int last = a[a.length-1];
            
            for(int j=a.length-1; j>0; j--){
                
                a[j] = a[j-1];
            }
            a[0] =last;
            
        }
        
        for(int b : a){
            System.out.print(b+" ");
        }
    }
}


3 4 5 1 2
