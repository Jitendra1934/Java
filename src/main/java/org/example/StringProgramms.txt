            if(count == 1 )

class Main {
    public static void main(String[] args) {
        
        String s = "Java css";
        
       String[] a = s.split(" ");
       
       String sum = "";
       
       for(int i=0; i<a.length; i++){
           
           sum = rev(a[i])+" "+sum;
       }
        System.out.println(sum);
    }
    
    public static String rev(String s ){
        
        String s1="";
        
        for(int i=0;i<s.length();i++){
            
            char ch = s.charAt(i);
            
            s1 = ch+s1;
        }
        
        return s1;
        
    }
}
