
-- git init -> Initializes the new git repository on current local directory




-- git config global --user.name <name> -> sets the name of author as your commits

-- git config global -user.email <email> -> sets the email of author as your commits





-- git branch <branch name> -> creates new branch 

-- git checkout <branch name> -> changes to different branch

-- git checkout -b <branch name> -> Creates the new branch and switches to it 

-- git branch -a -> list all the branches in local and remote 

-- git branch -d <branch name> -> delete the branch

-- git branch -m <old-branch-name> <new-branch-name> - Renames a local branch

-- git branch -D <branch name> -> delete the branch force fully 



-- git add <file name> -> stages the changes in working directory to be committed

-- git add . -> stages the all changes in working directory to be committed

-- git status -> display the status of working directory in staging area

-- git commit -m <commit message> -> commit the changes with a meaning full message 

-- git log -> Display the of all commits made in the repository 






-- git remote add <Url> adds a remote repository

-- git fetch -> fetch the changes from remote repository

-- git pull -> pull the changes from remote repository and merge the changes into current branch 

-- git push -> push the changes from current branch to repository





-- git merge -> Merges the changes one branch from another branch

-- git rebase -> when you are merging or combing the changes from one branch to another branch by moving the commits to tip of the target branch 

-- git reset <commit-hash> <file-name> -> Resets the current branch to a specific commit or file.

-- git reset --hard -> Resets the current branch to the last commit and discards all changes.

-- git reset --soft -> Resets the current branch to the last commit, but keeps changes in the staging area
